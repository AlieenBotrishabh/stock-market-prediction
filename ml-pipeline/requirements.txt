# Stock prediction pipeline.
#
# Deliberately lean so a GitHub Actions runner installs quickly:
#  - no yfinance (Yahoo is called directly with requests)
#  - no PyWavelets (Haar denoising is implemented in features.py)
#  - no statsmodels (baselines are naive-drift and linear-drift in numpy)
#  - no pyarrow (the on-disk cache is CSV)

requests>=2.31.0
numpy>=1.24.0,<2.0
pandas>=2.0.0

# LSTM. CPU-only is fine: a single-layer LSTM(150) over ~2,500 rows trains
# in a couple of minutes without a GPU.
tensorflow>=2.13.0

# Publishing predictions to MongoDB (optional; the pipeline runs without it)
pymongo>=4.6.0

python-dotenv>=1.0.0

# ONNX export, so the model can run inside the Node API at request time
# instead of only in a batch job. TensorFlow cannot fit in a Vercel
# function (~600MB vs a 250MB limit); the exported graph is ~390KB.
onnx>=1.16
tf2onnx>=1.17
onnxruntime>=1.17          # used to verify the export matches Keras

# Publishing bundles to the Hugging Face Hub
huggingface_hub>=0.23
